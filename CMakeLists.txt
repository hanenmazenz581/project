cmake_minimum_required(VERSION 3.21)
project(StudioCutPro VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets)
find_program(FFMPEG_EXECUTABLE ffmpeg)
qt_add_executable(StudioCutPro WIN32 src/main.cpp)
target_link_libraries(StudioCutPro PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
if(FFMPEG_EXECUTABLE)
  target_compile_definitions(StudioCutPro PRIVATE STUDIOCUT_FFMPEG_PATH="${FFMPEG_EXECUTABLE}")
endif()
if(MSVC)
  target_compile_options(StudioCutPro PRIVATE /W4 /permissive-)
else()
  target_compile_options(StudioCutPro PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS StudioCutPro RUNTIME DESTINATION .)
